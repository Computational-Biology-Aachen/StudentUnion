  21# Custom Latex Printer to handle units with negative integer exponents properly (Still need to be improved for more complex cases)
  22class RefinedUnitLatexPrinter(LatexPrinter):
  23    def _print_Pow(self, expr):
  24        # Custom handling for Unit objects with negative integer exponents
  25        if (
  26            isinstance(expr.base, Unit)
  27            and isinstance(expr.exp, Integer)
  28            and expr.exp < 0
  29        ):
  30            base_str = self._print(expr.base)
  31            exp_str = self._print(expr.exp)
  32            return f"{base_str}^{{{exp_str}}}"
  33        # Fallback to default behavior for other Pow expressions
  34        return super()._print_Pow(expr)
  35
  36    def _print_Mul(self, expr):
  37        # Separate terms with positive and negative exponents
  38        positive_terms = []
  39        negative_terms = []
  40
  41        for arg in expr.args:
  42            # Check if the argument is a Pow of a Unit with a negative integer exponent
  43            if (
  44                isinstance(arg, Pow)
  45                and isinstance(arg.base, Unit)
  46                and isinstance(arg.exp, Integer)
  47                and arg.exp < 0
  48            ):
  49                negative_terms.append(arg)
  50            else:
  51                positive_terms.append(arg)
  52
  53        # Print positive terms and negative terms
  54        # SymPy's default _print(Unit) for millimeter is \text{mm}, so no extra \cdot is needed within it.
  55        printed_positive_terms = [self._print(t) for t in positive_terms]
  56        printed_negative_terms = [self._print(t) for t in negative_terms]
  57
  58        all_printed_terms = []
  59        if printed_positive_terms:
  60            all_printed_terms.extend(printed_positive_terms)
  61        if printed_negative_terms:
  62            all_printed_terms.extend(printed_negative_terms)
  63
  64        if all_printed_terms:
  65            return r"\ ".join(all_printed_terms)
  66        else:
  67            # Fallback to default behavior if no specific handling is needed
  68            return super()._print_Mul(expr)

  75def calc_co2_conc(pco2: float, H_cp_co2: float = 3.4e-4):
  76    """Calculate the CO2 concentration based on CO2 partial pressure and Henry's law constant.
  77
  78    Args:
  79        pco2 (float): CO2 partial pressure [µbar].
  80        H_cp_co2 (float, optional): Henry's law constant for CO2 at 25°C [mM * Pa-1]. Defaults to 3.4e-4 (https://doi.org/10.5194/acp-23-10901-2023).
  81
  82    Returns:
  83        float: CO2 concentration in mM.
  84    """
  85    # Unit conversions
  86    H_cp_co2 = H_cp_co2 * 1e5  # [mM * bar-1]
  87    H_cp_co2 = H_cp_co2 * 1e-6  # [mM * µbar-1]
  88
  89    return H_cp_co2 * pco2

 105def _calc_ass(vc: float, gammastar: float, r_light: float, co2: float):
 106    """Calculate carbon assimilation based on the min-W approach, as introduced by Farquhar, von Caemmerer and Berry in 1980 and "reevaluated" by Lochoki and McGrath in 2025 (https://doi.org/10.1101/2025.03.11.642611).
 107
 108    Args:
 109        vc (float): Rubisco carboxylation rate [µmol m-2 s-1]
 110        gammastar (float): CO2 compensation point in the absence of non-photorespiratory CO2 release [µbar]
 111        r_light (float): Rate of non-photorespiratory CO2 release in the light [µmol m-2 s-1]
 112        co2 (float): CO2 partial pressure [µbar]
 113
 114    Returns:
 115        float: Net carbon assimilation rate [µmol m-2 s-1]
 116    """
 117    return vc * (1 - gammastar / co2) - r_light

 120def inject_fvcb(
 121    model: Model,
 122    co2: str | None = None,
 123    vc: str | None = None,
 124    pco2: str | None = None,
 125    H_cp_co2: str | None = None,
 126    gammastar: str | None = None,
 127    r_light: str | None = None,
 128    A: str | None = None,
 129) -> tuple[
 130    Model,
 131    str | None,
 132    str | None,
 133    str | None,
 134    str | None,
 135    str | None,
 136    str | None,
 137    str | None,
 138]:
 139    """Inject the FvCB model into a MxLpy model
 140
 141    Args:
 142        model (Model): MxLpy model to inject the FvCB model into
 143        co2 (str | None, optional): CO2 concentration name in model. Defaults to None.
 144        vc (str | None, optional): Rubisco carboxylation rate name in model. Defaults to None.
 145        pco2 (str | None, optional): CO2 partial pressure name in model. Defaults to None.
 146        H_cp_co2 (str | None, optional): Henry's law constant for CO2 name in model. Defaults to None.
 147        gammastar (str | None, optional): CO2 compensation point name in model. Defaults to None.
 148        r_light (str | None, optional): Rate of non-photorespiratory CO2 release in the light name in model. Defaults to None.
 149        A (str | None, optional): Net carbon assimilation rate name in model. Defaults to None.
 150
 151    Returns:
 152        tuple(model, co2, vc, pco2, H_cp_co2, gammastar, r_light, A): Injected model and variable names
 153    """
 154    # Copy inputted model to avoid modifying the original one
 155    model = copy.deepcopy(model)
 156
 157    # If either co2 or vc is None, return without modifications
 158    if co2 is None or vc is None or A is not None:
 159        return (model, co2, vc, pco2, H_cp_co2, gammastar, r_light, A)
 160
 161    # Check unit of vc, if in mM convert to µmol m-2 s-1 else assume in µmol m-2 s-1
 162    if model.get_raw_reactions()[vc].unit == (
 163        units.mmol / (units.liter * units.second)
 164    ):
 165        model.add_parameter(
 166            "Vstroma_factor", value=0.0112, unit=units.liter / units.sqm
 167        )
 168
 169        model.add_derived(
 170            vc + " µmol m-2 s-1",
 171            fn=mM_to_µmol_per_m2,
 172            args=[vc, "Vstroma_factor"],
 173            unit=units.mmol / (units.sqm * units.second),
 174        )
 175        vc = vc + " µmol m-2 s-1"
 176
 177    # Check what co2 is in model and get initial value
 178    if model.ids[co2] == "parameter":
 179        initial_val_co2 = model._parameters[co2].value
 180    elif model.ids[co2] == "variable":
 181        initial_val_co2 = model._variables[co2].initial_value
 182
 183    # If pco2 is None add it as parameter based on initial co2 value and H_cp_co2
 184    if pco2 is None:
 185        # If H_cp_co2 is None add it as parameter
 186        if H_cp_co2 is None:
 187            model.add_parameter("H_cp_co2", 3.4e-4, unit=units.mmol / units.pascal)
 188            H_cp_co2 = "H_cp_co2"
 189
 190        initial_val = initial_val_co2 / model._parameters[H_cp_co2].value
 191        model.add_parameter("pco2", initial_val, unit=units.micro * bar)
 192        pco2 = "pco2"
 193
 194        # Remove original co2 from model and add it as derived from pco2 and H_cp_co2
 195        model._remove_id(name=co2)
 196        model.add_derived(
 197            co2, fn=calc_co2_conc, args=[pco2, H_cp_co2], unit=units.mmol / units.liter
 198        )
 199
 200    if gammastar is None:
 201        model.add_parameter(
 202            gammastar := "gammastar",
 203            38.6,
 204            unit=units.micro * bar,
 205            source="https://doi.org/10.1101/2025.03.11.642611)",
 206        )
 207
 208    if r_light is None:
 209        model.add_parameter(
 210            r_light := "r_light",
 211            1.0,
 212            unit=units.mmol / (units.sqm * units.second),
 213            source="https://doi.org/10.1101/2025.03.11.642611",
 214        )
 215
 216    model.add_derived(
 217        A := "Assimilation",
 218        fn=_calc_ass,
 219        args=[vc, gammastar, r_light, pco2],
 220        unit=units.mmol / (units.sqm * units.second),
 221    )
 222
 223    return (model, co2, vc, pco2, H_cp_co2, gammastar, r_light, A)

 226def calculate_assimilation_minW(
 227    pco2: float,
 228    v_cmax: float = 100,
 229    km_co2: float = 259,
 230    o2: float = 210,
 231    km_o2: float = 179,
 232    j: float = 170,
 233    gammastar: float = 38.6,
 234    tp: float = 11.8,
 235    alpha_old: float = 0,
 236    r_light: float = 1,
 237) -> tuple[float, float, float, float]:
 238    """Calculate carbon assimilation based on the min-W approach.
 239
 240    Calculate carbon assimilation based on the min-W approach, as introduced by Farquhar, von Caemmerer and Berry in 1980 and "reevaluated" by Lochoki and McGrath in 2025 (https://doi.org/10.1101/2025.03.11.642611).
 241
 242    Args:
 243        pco2 (float): [µbar] Partial pressure of CO2. Normally chloroplastic CO2 partial pressure (Cc), but may be interchanged with intercellular CO2 partial pressure (Ci) under simplification of infinite mesophyll conductance
 244        v_cmax (float, optional): [µmol m-2 s-1] Maximum rate of Rubisco carboxylation activity. Defaults to 100.
 245        km_co2 (float, optional): [µbar] Michaelis-Menten constant for CO2. Defaults to 259.
 246        o2 (float, optional): [mbar] Partial pressure of O2 in the vicinity of Rubisco. Defaults to 210.
 247        km_o2 (float, optional): [mbar] Michaelis-Menten constant for O2. Defaults to 179.
 248        j (float, optional): [µmol m-2 s-1] Potential rate of linear electron transport going to support RuBP regeneration at a given light intensity. Defaults to 170.
 249        gammastar (float, optional): [µbar] CO2 compensation point in the absence of non-photorespiratory CO2 release. Defaults to 38.6.
 250        tp (float, optional): [µmol m-2 s-1] Potential rate of TPU. Defaults to 11.8.
 251        alpha_old (float, optional): [] Fraction of remaining glycolate carbon not returned to the chloroplast after accounting for carbon released as co2. Defaults to 0.
 252        r_light (float, optional): [µmol m-2 s-1] Rate of non photorespiratory CO2 release in the light. Defaults to 1.
 253
 254    Returns:
 255        A_n, wc, wj, wp: Carbon Assimilation rate and the three limiting rates
 256    """
 257    # Rubisco carboxylation limited rate
 258    wc = pco2 * v_cmax / (pco2 + km_co2 * (1 + o2 / km_o2))
 259    # RuBP regeneration limited rate
 260    wj = pco2 * j / (4 * pco2 + 8 * gammastar)
 261    # TPU limited rate
 262    if pco2 <= gammastar * (1 + 3 * alpha_old):
 263        wp = math.inf
 264    else:
 265        wp = 3 * pco2 * tp / (pco2 - gammastar * (1 + 3 * alpha_old))
 266
 267    # Net assimilation rate
 268    vc = min(wc, wj, wp)
 269    A_n = vc * (1 - gammastar / pco2) - r_light
 270
 271    return A_n, wc, wj, wp

 437def make_pam_protocol(
 438    pfd: str,
 439    length_period: float = 120,
 440    length_pulse: float = 0.8,
 441    pulse_intensity: float = 3000,
 442    actinic_light: float = 1000,
 443    dark_light: float = 40,
 444) -> tuple[pd.DataFrame, pd.DataFrame]:
 445    """Use make_protocol from mxlpy to create a PAM protocol
 446
 447    Create a PAM protocol using make_protocol from mxlpy. The protocol consists of 2 dark periods with a saturating pulse, followed by 10 light periods with a saturating pulse, followed by 10 dark periods of with a saturating pulse. The time units of the length_period and length_pulse should be the same as the model time unit.
 448
 449    Args:
 450        pfd (str): Name of PPFD parameter in model
 451        length_period (float, optional): Length between pulses in model time unit. Defaults to 120 s.
 452        length_pulse (float, optional): Length of the pulse in model time unit. Defaults to 0.8 s.
 453        pulse_intensity (float, optional): Intensity of the pulse in model PPFD unit. Defaults to 3000 µmol m-2 s-1.
 454        actinic_light (float, optional): Intensity of the actinic light in model PPFD unit. Defaults to 1000 µmol m-2 s-1.
 455        dark_light (float, optional): Intensity of the dark light in model PPFD unit. Defaults to 40 µmol m-2 s-1.
 456
 457    Returns:
 458        prtc (pd.DataFrame): PAM protocol as pd.DataFrame created using make_protocol from mxlpy
 459        shading (pd.DataFrame): Shading information for plotting the protocol as pd.DataFrame
 460    """
 461    length_nopulse = length_period - length_pulse
 462
 463    dark_period = [
 464        (length_nopulse, {pfd: dark_light}),
 465        (length_pulse, {pfd: pulse_intensity}),
 466    ]
 467    light_period = [
 468        (length_nopulse, {pfd: actinic_light}),
 469        (length_pulse, {pfd: pulse_intensity}),
 470    ]
 471
 472    prtc = make_protocol(dark_period * 2 + light_period * 10 + dark_period * 10)
 473
 474    shading = prtc[prtc[pfd] != pulse_intensity].copy()
 475    shading["Color"] = shading[pfd].apply(
 476        lambda x: "black" if x == dark_light else "white"
 477    )
 478
 479    return prtc, shading

 482def calc_pam_vals(
 483    fluo_result: pd.Series, peak_distance: float = 120
 484) -> tuple[pd.Series, pd.Series, pd.Series, pd.DataFrame]:
 485    """Calculate PAM values from fluorescence data.
 486
 487    Use the fluorescence data from a PAM protocol to calculate Fm, NPQ, Fmin, and the quantum yields Y(NO), Y(NPQ) and Y(II). To find the Fm values, the peaks in the fluorescence data are found using scipy.signal.find_peaks. The distance between the peaks should be the same length as a period used in the PAM protocol, however may need to be adjusted based on the fluorescence data. Bets to plot the Flourescence data and the calculated Fm to check if the peaks are found correctly.
 488
 489    Args:
 490        fluo_result (pd.Series): Fluorescence data as a pd.Series from mxlpy simulation.
 491        peak_distance (float, optional): Minimum distance between peaks, which should be the same length as a period used in the PAM protocol. However may need to be adjusted based on the fluorescence data. Defaults to 120 s.
 492
 493    Returns:
 494        Fm (pd.Series): Maximum fluorescence values
 495        NPQ (pd.Series): Non-photochemical quenching values
 496        Fmin (pd.Series): Minimum fluorescence values
 497        quant_yields (pd.DataFrame): Quantum yields (Y(NO), Y(NPQ), Y(II))
 498    """
 499
 500    # Find the indices of the Flourescence peaks (Fmaxs)
 501    peaks, _ = find_peaks(fluo_result, distance=peak_distance, height=0)
 502
 503    # Fm series
 504    Fm = fluo_result.iloc[peaks]
 505    Fm.name = "Flourescence Peaks (Fm)"
 506
 507    # Calculate NPQ
 508    NPQ = (Fm.iloc[0] - Fm) / Fm if len(Fm) > 0 else pd.Series(dtype=float)
 509    NPQ.name = "Non-Photochemical Quenching (NPQ)"
 510
 511    # Find the minima around the peaks
 512    prominences, prominences_left, prominences_right = peak_prominences(
 513        (fluo_result), peaks, wlen=peak_distance
 514    )
 515
 516    # Fmin is always the minima before the peak
 517    Fmin = fluo_result.iloc[prominences_left]
 518    Fmin.name = "Minimum Fluorescence before peaks (Fmin)"
 519
 520    # Quantum Yield of Non-Regulated Energy Loss (Y(NO))
 521    Y_NO = Fmin / Fm.iloc[0] if len(Fm) > 0 else pd.Series(dtype=float)
 522    Y_NO.name = "Y(NO)"
 523
 524    # Quantum Yield of Regulated Heat Dissipation (Y(NPQ))
 525    Y_NPQ = (
 526        Fmin / Fm.values - Fmin / Fm.iloc[0] if len(Fm) > 0 else pd.Series(dtype=float)
 527    )
 528    Y_NPQ.name = "Y(NPQ)"
 529
 530    # Quantum Yield of Photochemical Energy Conversion (Y(II))
 531    Y_II = (Fm.values - Fmin) / Fm.values if len(Fm) > 0 else pd.Series(dtype=float)
 532    Y_II.name = "Y(II)"
 533
 534    # pd.DataFrame of the three quantum yields
 535    quant_yields = pd.concat([Y_NO, Y_NPQ, Y_II], axis=1)
 536    quant_yields["Total"] = quant_yields.sum(axis=1)
 537
 538    return Fm, NPQ, Fmin, quant_yields

 546    """Create a PAM figure from a MxLpy model.

 549
 550    Args:

 554
 555    Returns:
 556        tuple[plt.Figure, tuple[plt.Axes, plt.Axes]]: A tuple containing the figure and the axes.
 557    """

270    A_n = vc * (1 - gammastar / pco2) - r_light

483def calc_pam_vals(
484    fluo_result: pd.Series, peak_distance: float = 120
485) -> tuple[pd.Series, pd.Series, pd.Series, pd.DataFrame]:
486    """Calculate PAM values from fluorescence data.
487
488    Use the fluorescence data from a PAM protocol to calculate Fm, NPQ, Fmin, and the quantum yields Y(NO), Y(NPQ) and Y(II). To find the Fm values, the peaks in the fluorescence data are found using scipy.signal.find_peaks. The distance between the peaks should be the same length as a period used in the PAM protocol, however may need to be adjusted based on the fluorescence data. Bets to plot the Flourescence data and the calculated Fm to check if the peaks are found correctly.
489
490    Args:
491        fluo_result (pd.Series): Fluorescence data as a pd.Series from mxlpy simulation.
492        peak_distance (float, optional): Minimum distance between peaks, which should be the same length as a period used in the PAM protocol. However may need to be adjusted based on the fluorescence data. Defaults to 120 s.
493
494    Returns:
495        Fm (pd.Series): Maximum fluorescence values
496        NPQ (pd.Series): Non-photochemical quenching values
497        Fmin (pd.Series): Minimum fluorescence values
498        quant_yields (pd.DataFrame): Quantum yields (Y(NO), Y(NPQ), Y(II))
499    """
500
501    # Find the indices of the Flourescence peaks (Fmaxs)
502    peaks, _ = find_peaks(fluo_result, distance=peak_distance, height=0)
503
504    # Fm series
505    Fm = fluo_result.iloc[peaks]
506    Fm.name = "Flourescence Peaks (Fm)"
507
508    # Calculate NPQ
509    NPQ = (Fm.iloc[0] - Fm) / Fm if len(Fm) > 0 else pd.Series(dtype=float)
510    NPQ.name = "Non-Photochemical Quenching (NPQ)"
511
512    # Find the minima around the peaks
513    prominences, prominences_left, prominences_right = peak_prominences(
514        (fluo_result), peaks, wlen=peak_distance
515    )
516
517    # Fmin is always the minima before the peak
518    Fmin = fluo_result.iloc[prominences_left]
519    Fmin.name = "Minimum Fluorescence before peaks (Fmin)"
520
521    # Quantum Yield of Non-Regulated Energy Loss (Y(NO))
522    Y_NO = Fmin / Fm.iloc[0] if len(Fm) > 0 else pd.Series(dtype=float)
523    Y_NO.name = "Y(NO)"
524
525    # Quantum Yield of Regulated Heat Dissipation (Y(NPQ))
526    Y_NPQ = (
527        Fmin / Fm.values - Fmin / Fm.iloc[0] if len(Fm) > 0 else pd.Series(dtype=float)
528    )
529    Y_NPQ.name = "Y(NPQ)"
530
531    # Quantum Yield of Photochemical Energy Conversion (Y(II))
532    Y_II = (Fm.values - Fmin) / Fm.values if len(Fm) > 0 else pd.Series(dtype=float)
533    Y_II.name = "Y(II)"
534
535    # pd.DataFrame of the three quantum yields
536    quant_yields = pd.concat([Y_NO, Y_NPQ, Y_II], axis=1)
537    quant_yields["Total"] = quant_yields.sum(axis=1)
538
539    return Fm, NPQ, Fmin, quant_yields
